Columns:
  id: Stable record id (never reused)
  jurisdiction: Two-letter state code, or US
  level: us, state, county or former-county
  year: 
  applications: Applications with an action of originated, approved not accepted or denied
  originations: Loans originated
  denials: Applications denied
  denial_rate: Denied / applications, percent
  originated_dollars: Dollar amount of loans originated
  applications_purchase: 
  originations_purchase: 
  denials_purchase: 
  applications_refinance: 
  originations_refinance: 
  denials_refinance: 
  applications_cash_out_refinance: 
  originations_cash_out_refinance: 
  denials_cash_out_refinance: 
  text: The record in one plain-English sentence
  lastChecked: The day it was last confirmed (YYYY-MM-DD)
  confidence: high, medium or low: computed from status, a second check, age against the refresh cadence and sampling error (rule: /data/#confidence)
  confidenceBasis: The reasons for the confidence level, in a short phrase
  county: County FIPS code as HMDA reports it (in mortgage-lending-values.csv, as id, column and value rows)
  countyName:  (in mortgage-lending-values.csv, as id, column and value rows)

  priorYear:  (in mortgage-lending-values.csv, as id, column and value rows)
